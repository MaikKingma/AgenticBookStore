#!/usr/bin/env bash
# Fresh sandbox (arg 1): simulate-level-0.sh level-0-start, then build-level.sh level-4-agentos; compare with sandbox 1.
set -eu
. "$COMMS_TMP/impl-t13-lib.sh"
iso_env
SBX2=${1:-$SP/s10-t13-impl-sbx-2}
rm -rf "$SBX2"
export GIT_AUTHOR_NAME=T13 GIT_AUTHOR_EMAIL=t13@localhost GIT_COMMITTER_NAME=T13 GIT_COMMITTER_EMAIL=t13@localhost
echo "== simulate-level-0.sh level-0-start"
bash "$S/scripts/simulate-level-0.sh" --repo "$WT" level-0-start "$SBX2" 2>&1 | grep -v '^strip-path' | tail -5
echo "== build-level.sh level-4-agentos"
bash "$S/scripts/build-level.sh" level-4-agentos "$SBX2" 2>&1 | grep -E 'build-level|committed|error' | tail -12
echo "== commits"
git -C "$SBX2" log --format='%h %s'
echo "== trees"
echo "fresh build HEAD tree: $(git -C "$SBX2" rev-parse 'HEAD^{tree}')"
echo "sandbox 1 (hand run)    HEAD tree: $(git -C "$SBX1" rev-parse 'HEAD^{tree}')"
echo "level-3 tree inside the fresh build (HEAD~1): $(git -C "$SBX2" rev-parse 'HEAD~1^{tree}')"
if [ "$(git -C "$SBX2" rev-parse 'HEAD^{tree}')" = "$(git -C "$SBX1" rev-parse 'HEAD^{tree}')" ]; then echo "RESULT trees equal"; else echo "RESULT trees DIFFER"; git -C "$SBX1" diff --stat HEAD "$(git -C "$SBX2" rev-parse HEAD)" 2>&1 | tail -5; fi
echo "== diff HEAD~1 HEAD of the fresh build against the manifest targets"
git -C "$SBX2" -c core.quotePath=false diff --name-only HEAD~1 HEAD | LC_ALL=C sort > "$RAW/sbx2-changed.txt"
awk -F '\t' '/^#/ {next} NF {print $4}' "$S/level-4-agentos/manifest.tsv" | LC_ALL=C sort > "$RAW/manifest-targets.txt"
if cmp -s "$RAW/sbx2-changed.txt" "$RAW/manifest-targets.txt"; then echo "RESULT commit changes exactly the $(wc -l < "$RAW/manifest-targets.txt" | tr -d ' ') manifest targets"; else echo "RESULT MISMATCH"; diff "$RAW/sbx2-changed.txt" "$RAW/manifest-targets.txt"; fi
echo "== working tree status of the fresh build"
git -C "$SBX2" status --porcelain | wc -l | sed 's/^/dirty entries: /'
