#!/usr/bin/env bash
# Stage every generated and written file of sandbox 1 as an `add` row of S/level-4-agentos, then commit sandbox 1.
set -eu
. "$COMMS_TMP/impl-t13-lib.sh"
iso_env
STEP=$S/level-4-agentos
rm -rf "$STEP"
mkdir -p "$STEP/files"
encode() { # leading dot of a path component becomes dot-
  local out="" comp
  local IFS=/
  for comp in $1; do
    case $comp in .*) comp="dot-${comp#.}" ;; esac
    out="${out:+$out/}$comp"
  done
  printf '%s' "$out"
}
git -C "$SBX1" ls-files --others --exclude-standard | LC_ALL=C sort > "$RAW/stage-list.txt"
echo "untracked files in sandbox 1: $(wc -l < "$RAW/stage-list.txt" | tr -d ' ')"
echo "ignored files in sandbox 1 (must be none): $(git -C "$SBX1" status --porcelain --ignored | grep -c '^!!' || true)"
printf '# op\tgroup\tstaged\ttarget\tmode\tbase\n' > "$STEP/manifest.tsv"
while IFS= read -r f; do
  enc=$(encode "$f")
  mkdir -p "$STEP/files/$(dirname "$enc")"
  cp "$SBX1/$f" "$STEP/files/$enc"
  printf 'add\tconfig\tfiles/%s\t%s\t644\t-\n' "$enc" "$f" >> "$STEP/manifest.tsv"
done < "$RAW/stage-list.txt"
echo "manifest rows: $(grep -vc '^#' "$STEP/manifest.tsv")"
export GIT_AUTHOR_NAME=T13 GIT_AUTHOR_EMAIL=t13@localhost GIT_COMMITTER_NAME=T13 GIT_COMMITTER_EMAIL=t13@localhost
. "$S/scripts/lib-steps.sh"
git -C "$SBX1" add --all
git -C "$SBX1" commit --quiet -m "$(step_message level-4-agentos)"
echo "sandbox 1 commit: $(git -C "$SBX1" log -1 --format='%h %s')"
echo "sandbox 1 tree: $(git -C "$SBX1" rev-parse 'HEAD^{tree}')"
git -C "$SBX1" -c core.quotePath=false diff --name-status HEAD~1 HEAD | LC_ALL=C sort -k2 > "$RAW/sbx1-diff.txt"
cat "$RAW/sbx1-diff.txt" | mask
