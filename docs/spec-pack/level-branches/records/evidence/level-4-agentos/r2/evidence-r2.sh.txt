#!/usr/bin/env bash
# Closing round r2 evidence (CR-2, CR-3, CR-6, CR-7): fresh trees of all three tracks, self-test, checklist hunk map,
# literal check with controls, claims with the derived C32, hygiene. Output: evidence/level-4-agentos/r2/.
set -u
. "$COMMS_TMP/impl-t13-lib.sh"
iso_env
EV2=$S/records/evidence/level-4-agentos/r2
mkdir -p "$EV2"
fin() { ascii | mask; }
CT=$COMMS_TMP
REC=$S/records
export GIT_AUTHOR_NAME=T13 GIT_AUTHOR_EMAIL=t13@localhost GIT_COMMITTER_NAME=T13 GIT_COMMITTER_EMAIL=t13@localhost

# ---- CR-7: fresh trees of the three tracks on the merged tree
{
  echo "# Fresh simulate-level-0.sh level-0-start plus build-level.sh <step> for each track, on the merged T13 tree (spec tip 9089969 merged as ca66a3b)."
  echo "# worktree HEAD: $(git -C "$WT" log -1 --format='%h %s' | cut -c1-90)"
  for pair in "level-4-openspec:1c279415:7" "level-4-speckit:c9d758b9:8" "level-4-agentos:0a55b23f:9"; do
    step=${pair%%:*}; rest=${pair#*:}; want=${rest%%:*}; n=${rest##*:}
    D=$SP/s10-t13-impl-sbx-$n
    rm -rf "$D"
    bash "$S/scripts/simulate-level-0.sh" --repo "$WT" level-0-start "$D" > /dev/null 2>&1
    bash "$S/scripts/build-level.sh" "$step" "$D" > "$RAW/r2-build-$step.txt" 2>&1
    got=$(git -C "$D" rev-parse 'HEAD^{tree}')
    rows=$(awk -F '\t' '/^#/ {next} NF {c++} END {print c+0}' "$S/$step/manifest.tsv")
    changed=$(git -C "$D" diff --name-only HEAD~1 HEAD | wc -l | tr -d ' ')
    [ "${got#$want}" != "$got" ] && st="equal to the expected $want" || st="DIFFERENT from the expected $want"
    echo "$step: tree $got ($st); manifest rows $rows, files changed by the step commit $changed; commits: $(git -C "$D" rev-list --count HEAD)"
    grep -E '^build-level: committed' "$RAW/r2-build-$step.txt" | tail -1 | sed 's/^/  /'
  done
} 2>&1 | fin > "$EV2/fresh-trees.txt"
{
  echo "# bash S/tests/test-staging.sh on the merged tree (TMPDIR under \$SP, HR1). Last 6 lines:"
  bash "$S/tests/test-staging.sh" > "$RAW/r2-test-staging.txt" 2>&1
  echo "exit code: $?"
  tail -6 "$RAW/r2-test-staging.txt"
} 2>&1 | fin > "$EV2/staging-selftest.txt"

# ---- claims with the derived C32 (CR-6)
SBX_OVERRIDE=$SP/s10-t13-impl-sbx-4 bash "$CT/impl-t13-claims.sh" 2>&1 | fin > "$EV2/claim-searches.txt"

# ---- CR-2: the checklist diff against the spec tip, hunk by hunk
K=$REC/rehearsal-checklist.md
{
  echo "# rehearsal-checklist.md against the spec tip pre-edit copy (the merged tree before this round): diff -U0 hunk map. Expected: three FU-T10-1 hunks and one insertion after the Level 3 section."
  diff -U0 "$SP/s10-t13-impl-r2-rehearsal-checklist.orig.txt" "$K" | grep -E '^@@'
  echo "# the three one-line hunks, new text (first 330 characters):"
  for n in 402 463 543; do printf 'line %s: ' "$n"; sed -n "${n}p" "$K" | grep -o -E 'level-3.md. Z4 lists the four cases of the commit hook \(the fifth row is the Stop handler\)\.|redo R3-1 setup steps 2 and 3|variants C, D and E of R3-1'; done
  echo "# bytes before the insertion hunk equal the pre-edit bytes except the three hunks: lines changed in the first 547 lines: $(diff "$SP/s10-t13-impl-r2-rehearsal-checklist.orig.txt" <(head -547 "$K") | grep -c '^>')"
  echo "# the old texts are gone: 'Z4 lists all four cases' $(grep -c 'Z4 lists all four cases' "$K"), 'setup step 2 (the failing' $(grep -c 'setup step 2 (the failing' "$K"), 'variants C and D of R3-1' $(grep -c 'variants C and D of R3-1' "$K")"
  echo "# Level 3 and earlier text: first 547 lines vs the pre-edit copy, differing lines: $(diff <(head -547 "$SP/s10-t13-impl-r2-rehearsal-checklist.orig.txt") <(head -547 "$K") | grep -c '^<')"
} 2>&1 | fin > "$EV2/checklist-hunks.txt"

# ---- CR-3: pointers resolve by grep -F, literals are found in their records
{
  echo "# Level 4 section of rehearsal-checklist.md (CR-3). Script: checklist-literals.py.txt."
  echo "# 1. every pointer heading resolves by grep -F in its record (one line each; 12 rows):"
  python3 -I - "$K" "$REC" <<'PY'
import re, subprocess, sys
k, rec = sys.argv[1:3]
t = open(k).read()
t = t[t.index("\n## Level 4\n"):]
n = 0
for row in t.split("\n"):
    c = [x.strip() for x in row.strip().strip("|").split("|")]
    if len(c) == 4 and re.match(r"(R4A|RR-S|R4C)", c[0]):
        f = c[1].strip("`"); h = c[2]
        r = subprocess.run(["grep", "-c", "-F", "### " + h, rec + "/" + f], capture_output=True, text=True)
        n += 1
        print("%-6s %-20s grep -F count %s  | %s" % (c[0], f, r.stdout.strip(), h[:60]))
print("pointer rows: %d" % n)
PY
  echo "# 2. every backticked literal of the section is found on one line of its record (per-track rows: that track's record):"
  python3 -I "$CT/impl-t13-literals.py" "$REC"
  echo "# control: one literal broken in memory (git status --short becomes git status --shortX) must be reported"
  python3 -I "$CT/impl-t13-literals.py" "$REC" --misspell | tail -3
  echo "# 3. branch names in the section: $(sed -n '/^## Level 4$/,$p' "$K" | grep -o -E 'start/level-4-[a-z]+' | sort | uniq -c | tr -s ' ' | tr '\n' ';')  step_branch prints: $(bash -c ". '$S/scripts/lib-steps.sh'; for s in level-4-openspec level-4-speckit level-4-agentos; do step_branch \$s; done" | tr '\n' ' ')"
  echo "# 4. pack words (spec 10 lines 478 to 481) in the three rows: GIVEN / WHEN / THEN quotes present in each: $(sed -n '/^## Level 4$/,$p' "$K" | grep -c 'a fresh clone on the track.s start branch'), $(sed -n '/^## Level 4$/,$p' "$K" | grep -c 'a$'); the pack lines:"
  sed -n 478,481p "$S/../specs/10-harness-reference-solutions.md"
  echo "# the quoted pack words are substrings of the pack text (whitespace joined): "
  python3 -I - "$S/../specs/10-harness-reference-solutions.md" "$K" <<'PY'
import re, sys
pack = " ".join(open(sys.argv[1]).read().split("\n")[477:481])
pack = re.sub(r"\s+", " ", pack)
t = open(sys.argv[2]).read()
t = t[t.index("\n## Level 4\n"):]
t = re.sub(r"\s+", " ", t)
for w in ["a fresh clone on the track's start branch", "a facilitator who has not used the framework before ships Level 4 brief A following `workshop/sdd-framework-guide.md`", "the feature is committed through `/ship` with verify green within 35 minutes (record the time per track)"]:
    print("in the pack: %s, in the section %d times: %s" % (w in pack, t.count(w), w[:50]))
PY
} 2>&1 | fin > "$EV2/checklist-level4-checks.txt"
for pair in "checklist-literals.py:impl-t13-literals.py" "evidence-r2.sh:impl-t13-r2.sh" "claim-searches.sh:impl-t13-claims.sh"; do
  out=${pair%%:*}; src=${pair##*:}
  sed -e "s|$COMMS_TMP|\$COMMS_TMP|g" -e "s|$SP|\$SP|g" -e "s|$REALHOME|\$REALHOME|g" "$CT/$src" | LC_ALL=C perl -CS -pe 's/[^\x00-\x7F]/?/g' > "$EV2/$out.txt"
done
echo "r2 evidence written: $(ls "$EV2" | wc -l) files"
